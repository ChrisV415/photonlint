### Part 1: The Engine (Python + `gdstk` + `shapely`)

While `gdstk` is incredibly fast for parsing GDSII, doing complex geometric distance math (like spacing and bend radius on fractured polygons) from scratch is a nightmare. The industry secret is to use **`gdstk` for parsing** and **`shapely` for the geometric math**.

**2. Create `drc_engine.py`:**
```python
import gdstk
import math
from shapely.geometry import Polygon as ShapelyPolygon

class PhotonLintEngine:
    def __init__(self, min_width_um=0.15, min_spacing_um=0.15, min_bend_radius_um=5.0):
        self.min_width = min_width_um
        self.min_spacing = min_spacing_um
        self.min_bend_radius = min_bend_radius_um
        self.violations = []

    def load_gds(self, filepath, layer=1, datatype=0):
        """Parses GDSII and extracts polygons from the target waveguide layer."""
        library = gdstk.read_gds(filepath)
        top_cells = library.top_level()
        if not top_cells:
            raise ValueError("No top-level cell found in GDS file.")
        
        polygons = []
        for cell in top_cells:
            for poly in cell.polygons:
                if poly.layer == layer and poly.datatype == datatype:
                    polygons.append(poly)
        return polygons

    def check_spacing(self, gdstk_polygons):
        """Checks minimum spacing between waveguides using Shapely."""
        # Convert gdstk polygons to shapely polygons for geometric operations
        shapely_polys = []
        for p in gdstk_polygons:
            try:
                shapely_polys.append(ShapelyPolygon(p.points).buffer(0)) # buffer(0) fixes self-intersections
            except Exception:
                continue
        
        # O(N^2) check. Fine for MVP, will need spatial indexing (R-tree) for Enterprise.
        for i in range(len(shapely_polys)):
            for j in range(i + 1, len(shapely_polys)):
                dist = shapely_polys[i].distance(shapely_polys[j])
                if dist < self.min_spacing:
                    self.violations.append({
                        "rule": "Minimum Spacing",
                        "required": f"{self.min_spacing} µm",
                        "actual": f"{dist:.3f} µm",
                        "severity": "CRITICAL"
                    })
                    return # Stop at first violation for MVP speed

    def check_bend_radius(self, gdstk_polygons):
        """Heuristic check for sharp bends by analyzing angles between fractured polygon edges."""
        for poly in gdstk_polygons:
            points = poly.points
            if len(points) < 3:
                continue
            
            for i in range(len(points) - 2):
                p1, p2, p3 = points[i], points[i+1], points[i+2]
                
                # Calculate vectors
                v1 = (p2[0] - p1[0], p2[1] - p1[1])
                v2 = (p3[0] - p2[0], p3[1] - p2[1])
                
                mag1 = math.hypot(v1[0], v1[1])
                mag2 = math.hypot(v2[0], v2[1])
                
                if mag1 == 0 or mag2 == 0:
                    continue
                    
                # Calculate angle between consecutive segments
                cos_theta = (v1[0]*v2[0] + v1[1]*v2[1]) / (mag1 * mag2)
                cos_theta = max(-1.0, min(1.0, cos_theta)) # Clamp for acos
                angle_rad = math.acos(cos_theta)
                
                # If the angle changes sharply (>10 deg) over a very short segment, it's a sharp bend
                if angle_rad > math.radians(10) and mag1 < self.min_bend_radius:
                    self.violations.append({
                        "rule": "Minimum Bend Radius",
                        "required": f"{self.min_bend_radius} µm",
                        "actual": f"Sharp bend (seg: {mag1:.2f} µm, angle: {math.degrees(angle_rad):.1f}°)",
                        "severity": "WARNING"
                    })
                    return

    def run_drc(self, filepath, layer=1, datatype=0):
        self.violations = []
        polygons = self.load_gds(filepath, layer, datatype)
        
        if not polygons:
            return {"status": "ERROR", "message": "No polygons found on the specified waveguide layer."}
            
        self.check_spacing(polygons)
        self.check_bend_radius(polygons)
        
        if not self.violations:
            return {"status": "PASS", "message": "Pre-flight DRC check passed."}
        else:
            return {"status": "FAIL", "violations": self.violations}
```

**3. Create `main.py` (The API):**
```python
from fastapi import FastAPI, UploadFile, File
from drc_engine import PhotonLintEngine
import tempfile
import os

app = FastAPI(title="PhotonLint API")

@app.post("/run-drc")
async def run_drc(file: UploadFile = File(...)):
    # Save uploaded file temporarily
    with tempfile.NamedTemporaryFile(delete=False, suffix=".gds") as tmp:
        tmp.write(await file.read())
        tmp_path = tmp.name

    try:
        engine = PhotonLintEngine(min_spacing_um=0.15, min_bend_radius_um=5.0)
        result = engine.run_drc(tmp_path)
        return result
    finally:
        os.remove(tmp_path)
```
*Run it with: `uvicorn main:app --reload`*
