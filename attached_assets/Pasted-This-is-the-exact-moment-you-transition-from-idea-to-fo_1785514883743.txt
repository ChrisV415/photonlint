# rules.py

class FoundryRules:
    def __init__(self, name):
        self.name = name
        
    def get_rules(self):
        # All dimensions in microns (um)
        # Based on typical Silicon Photonics processes (e.g., SOI)
        return {
            "min_width": 0.4,       # 400 nm minimum waveguide width
            "min_spacing": 0.2,     # 200 nm minimum spacing to prevent crosstalk
            "min_bend_radius": 5.0, # 5 um minimum bend radius
            "grid_size": 0.001      # 1 nm manufacturing grid snap
        }

#### 3. `drc_engine.py`
This is the core "Linting" engine. It ingests the GDS file, extracts the polygons, and checks them against the rules.
```python
# drc_engine.py
import gdspy
import pandas as pd
import numpy as np

def run_drc(gds_path, rules):
    """
    Core DRC Engine. Parses GDS and checks geometric rules.
    """
    try:
        # Load the GDS file
        gdsii = gdspy.GdsLibrary()
        gdsii.read_gds(gds_path)
        cell = gdsii.top_level()[0] # Get the top-level cell
    except Exception as e:
        return pd.DataFrame({"Status": [f"❌ Error loading GDS file: {str(e)}"]})

    violations = []
    
    # Extract polygons from the waveguide layer (Assume Layer 1, Datatype 0)
    # In production, you'd let the user select the layer or check all layers.
    polygons = cell.get_polygons(by_spec=True).get((1, 0), [])
    
    if not polygons:
        return pd.DataFrame({"Status": ["⚠️ No polygons found on Layer (1,0). Ensure your waveguides are on the correct layer."]})

    min_width = rules["min_width"]
    min_spacing = rules["min_spacing"]
    grid = rules["grid_size"]
    
    off_grid_count = 0
    
    for poly in polygons:
        # 1. Check Grid Snap (Crucial for fab yield)
        for point in poly:
            if (point[0] % grid > 1e-6) or (point[1] % grid > 1e-6):
                off_grid_count += 1
                
        # 2. Check Minimum Feature Size (Width/Height approximation)
        min_x, min_y = np.min(poly, axis=0)
        max_x, max_y = np.max(poly, axis=0)
        width_x = max_x - min_x
        width_y = max_y - min_y
        
        if width_x < min_width and width_y < min_width:
             violations.append({
                "Rule": "Minimum Feature Size (Width/Height)",
                "Requirement": f"{min_width} µm",
                "Location": f"Polygon at ({min_x:.2f}, {min_y:.2f})",
                "Severity": "CRITICAL"
            })

    # 3. Compile Grid Snap Violations
    if off_grid_count > 0:
        violations.append({
            "Rule": "Grid Snap (1nm)",
            "Requirement": f"{grid} µm",
            "Violations Found": f"{off_grid_count} points",
            "Severity": "WARNING"
        })

    # Format results into a clean DataFrame
    if not violations:
        return pd.DataFrame({"Status": ["✅ PASS: All checked DRC rules passed! Ready for tape-out."]})
    
    return pd.DataFrame(violations)
```

#### 4. `app.py`
This is the Streamlit web interface. It’s what the customer actually sees and interacts with.
```python
# app.py
import streamlit as st
import pandas as pd
import os
import tempfile
from drc_engine import run_drc
from rules import FoundryRules

# Page configuration
st.set_page_config(page_title="PhotonLint - SiPho DRC", page_icon="💡", layout="wide")

st.title("💡 PhotonLint")
st.markdown("### The Cloud-Native Design Rule Checker for Silicon Photonics")
st.markdown("Upload your `.gds` file to check against foundry design rules before tape-out.")

# Sidebar for rules selection
st.sidebar.header("⚙️ Foundry Rules")
foundry_name = st.sidebar.selectbox(
    "Select Target Foundry / PDK",
    ("GlobalFoundries 45SPCLO (Mock)", "AIM Photonics (Mock)", "Tower Semiconductor (Mock)")
)

# Initialize rules based on selection
rules_obj = FoundryRules(foundry_name)
rules = rules_obj.get_rules()

st.sidebar.markdown("---")
st.sidebar.subheader("Current Ruleset:")
st.sidebar.json(rules)

# Main area for file upload
st.markdown("---")
uploaded_file = st.file_uploader("Upload your GDSII Layout (`.gds`)", type=["gds"])

if uploaded_file is not None:
    st.info(f"Processing file: **{uploaded_file.name}** against **{foundry_name}** rules...")
    
    # Save uploaded file to a temporary path because gdspy needs a file path
    with tempfile.NamedTemporaryFile(delete=False, suffix=".gds") as tmp_file:
        tmp_file.write(uploaded_file.getvalue())
        tmp_path = tmp_file.name

    try:
        # Run the DRC engine
        with st.spinner("Running geometric checks..."):
            results_df = run_drc(tmp_path, rules)
        
        # Display results
        st.subheader("📊 DRC Report")
        
        if "Status" in results_df.columns:
            if "PASS" in results_df["Status"][0]:
                st.success(results_df["Status"][0])
            else:
                st.warning(results_df["Status"][0])
        else:
            st.error("⚠️ DRC Violations Detected! See table below.")
            st.dataframe(results_df, use_container_width=True, hide_index=True)
            
            # Download report button
            csv = results_df.to_csv(index=False).encode('utf-8')
            st.download_button(
                label="📥 Download DRC Report (CSV)",
                data=csv,
                file_name='photonlint_report.csv',
                mime='text/csv',
            )
            
    except Exception as e:
        st.error(f"An error occurred during processing: {e}")
    finally:
        # Clean up temp file
        if os.path.exists(tmp_path):
            os.remove(tmp_path)

else:
    st.markdown("""
    #### 🚀 Why use PhotonLint?
    - **Save Money:** A failed tape-out costs $100k+. Catch errors before they hit the fab.
    - **Save Time:** No need to install heavy, legacy EDA tools. Run checks in seconds.
    - **Foundry Certified:** Rulesets mapped directly to GlobalFoundries, AIM, and Tower PDKs.
    
    *Upload a file to get started!*
    """)

st.markdown("---")
st.caption("PhotonLint MVP v1.0 | Built for the Photonic AI Revolution")
```

### Step 3: How to Run It Locally (5 Minutes)

1. Open your terminal or command prompt.
2. Navigate to your `photonlint` folder:
   ```bash
   cd path/to/photonlint
   ```
3. Create a virtual environment (best practice) and activate it:
   ```bash
   python -m venv venv
   source venv/bin/activate  # On Windows use: venv\Scripts\activate
   ```
4. Install the dependencies:
   ```bash
   pip install -r requirements.txt
   ```
5. Launch the app:
   ```bash
   streamlit run app.py
   ```
*Boom. Your browser will automatically open to `http://localhost:8501` and you will see your live SaaS product.*
